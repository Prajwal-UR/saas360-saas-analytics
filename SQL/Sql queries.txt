USE saas360;

SHOW TABLES;
SELECT COUNT(*) FROM accounts_clean;
SELECT COUNT(*) FROM subscriptions;
SELECT COUNT(*) FROM feature_usage;
SELECT COUNT(*) FROM support_tickets;
SELECT COUNT(*) FROM churn_events;

SELECT
    a.account_id,
    a.account_name,
    a.industry,
    a.plan_tier,
    SUM(s.mrr_amount) AS total_mrr,
    SUM(s.arr_amount) AS total_arr
FROM accounts_clean a
JOIN subscriptions_clean s
    ON a.account_id = s.account_id
GROUP BY
    a.account_id,
    a.account_name,
    a.industry,
    a.plan_tier
ORDER BY total_mrr DESC;

SELECT
    plan_tier,
    COUNT(DISTINCT account_id) AS customers,
    SUM(mrr_amount) AS total_mrr,
    SUM(arr_amount) AS total_arr,
    AVG(mrr_amount) AS avg_mrr
FROM subscriptions_clean
GROUP BY plan_tier
ORDER BY total_mrr DESC;

SELECT
    plan_tier,
    COUNT(*) AS total_customers,
    SUM(churn_flag) AS churned_customers,
    ROUND(
        SUM(churn_flag) * 100.0 / COUNT(*),
        2
    ) AS churn_rate_pct
FROM accounts_clean
GROUP BY plan_tier
ORDER BY churn_rate_pct DESC;

SELECT
    churn_flag,
    COUNT(*) AS count
FROM accounts_clean
GROUP BY churn_flag;

SELECT
    plan_tier,
    COUNT(*) AS total_customers,
    SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(
        SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) * 100.0
        / COUNT(*),
        2
    ) AS churn_rate_pct
FROM accounts_clean
GROUP BY plan_tier
ORDER BY churn_rate_pct DESC;

SELECT
    industry,
    COUNT(*) AS total_customers,
    SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(
        SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) * 100.0
        / COUNT(*),
        2
    ) AS churn_rate_pct
FROM accounts_clean
GROUP BY industry
ORDER BY churn_rate_pct DESC;

SELECT
    referral_source,
    COUNT(*) AS total_customers,
    SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) AS churned_customers,
    ROUND(
        SUM(CASE WHEN churn_flag = 'True' THEN 1 ELSE 0 END) * 100.0
        / COUNT(*),
        2
    ) AS churn_rate_pct
FROM accounts_clean
GROUP BY referral_source
ORDER BY churn_rate_pct DESC;

SELECT
    CASE
        WHEN a.churn_flag = 'True' THEN 'Churned'
        ELSE 'Retained'
    END AS customer_status,
    COUNT(DISTINCT a.account_id) AS customers,
    ROUND(SUM(s.mrr_amount), 2) AS total_mrr,
    ROUND(SUM(s.arr_amount), 2) AS total_arr,
    ROUND(AVG(s.mrr_amount), 2) AS avg_mrr
FROM accounts_clean a
JOIN subscriptions_clean s
    ON a.account_id = s.account_id
GROUP BY
    CASE
        WHEN a.churn_flag = 'True' THEN 'Churned'
        ELSE 'Retained'
    END
ORDER BY total_mrr DESC;


WITH customer_revenue AS (
    SELECT
        a.account_id,
        a.account_name,
        a.plan_tier,
        SUM(s.mrr_amount) AS total_mrr
    FROM accounts_clean a
    JOIN subscriptions_clean s
        ON a.account_id = s.account_id
    GROUP BY
        a.account_id,
        a.account_name,
        a.plan_tier
)

SELECT
    account_id,
    account_name,
    plan_tier,
    ROUND(total_mrr, 2) AS total_mrr
FROM customer_revenue
ORDER BY total_mrr DESC
LIMIT 10;


WITH customer_revenue AS (
    SELECT
        a.account_id,
        a.account_name,
        a.plan_tier,
        SUM(s.mrr_amount) AS total_mrr
    FROM accounts_clean a
    JOIN subscriptions_clean s
        ON a.account_id = s.account_id
    GROUP BY
        a.account_id,
        a.account_name,
        a.plan_tier
)

SELECT
    account_id,
    account_name,
    plan_tier,
    ROUND(total_mrr, 2) AS total_mrr,
    DENSE_RANK() OVER (
        ORDER BY total_mrr DESC
    ) AS revenue_rank
FROM customer_revenue
ORDER BY revenue_rank;


SELECT
    a.industry,
    COUNT(DISTINCT a.account_id) AS total_customers,

    SUM(
        CASE
            WHEN a.churn_flag = 'True' THEN 1
            ELSE 0
        END
    ) AS churned_customers,

    ROUND(
        SUM(
            CASE
                WHEN a.churn_flag = 'True' THEN 1
                ELSE 0
            END
        ) * 100.0 / COUNT(DISTINCT a.account_id),
        2
    ) AS churn_rate_pct,

    ROUND(
        SUM(
            CASE
                WHEN a.churn_flag = 'True'
                THEN s.mrr_amount
                ELSE 0
            END
        ),
        2
    ) AS churned_customer_mrr

FROM accounts_clean a
JOIN subscriptions_clean s
    ON a.account_id = s.account_id

GROUP BY a.industry
ORDER BY churn_rate_pct DESC;
